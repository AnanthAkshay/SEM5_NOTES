X = wine[ [ "fixed acidity", "volatile acidity", "citric acid", "residual sugar", "chlorides", "free sulfur dioxide", "total sulfur dioxide", "density", "pH", "sulphates", "alcohol" ] ] #SELECT TARGET VARIABLE y = wine["quality"] X_train, X_test, y_train, y_test = train_test_split(X,y,test_size=0.20,random_state=42)
# training and testing data print("Training samples:", len(X_train)) print("Testing samples:", len(X_test))
Training samples: 5197
Testing samples: 1300
# STANDARDIZE THE FEATURES scaler = StandardScaler() X_train = scaler.fit_transform(X_train) X_test = scaler.transform(X_test)
# LINEAR REGRESSION linear_model = LinearRegression() # Train model linear_model.fit(X_train, y_train) # Make predictions linear_prediction = linear_model.predict(X_test) # Calculate metrics r2_linear = r2_score(y_test,linear_prediction) rmse_linear = mean_squared_error(y_test,linear_prediction) ** 0.5 mae_linear = mean_absolute_error(y_test,linear_prediction) print("R2 Score :", round(r2_linear, 4)) print("RMSE :", round(rmse_linear, 4)) print("MAE :", round(mae_linear, 4)) 
R2 Score : 0.2598

# RIDGE REGRESSION ridge_model = Ridge(alpha=1) # Train model ridge_model.fit(X_train, y_train) # Make predictions ridge_prediction = ridge_model.predict(X_test) # Calculate metrics r2_ridge = r2_score(y_test,ridge_prediction) rmse_ridge = mean_squared_error(y_test,ridge_prediction) ** 0.5 mae_ridge = mean_absolute_error(y_test,ridge_prediction) print("R2 Score :", round(r2_ridge, 4)) print("RMSE :", round(rmse_ridge, 4)) print("MAE :", round(mae_ridge, 4))
R2 Score : 0.2598

# COMPARE BOTH MODELS results = pd.DataFrame({ "Model": ["Linear Regression","Ridge Regression"], "R2": [r2_linear,r2_ridge], "RMSE": [rmse_linear,rmse_ridge], "MAE": [mae_linear,mae_ridge] })
print("MODEL COMPARISON") print(results.round(4)) results.plot(x="Model",y=["R2", "RMSE", "MAE"],kind="bar",figsize=(10, 6)) plt.title("Linear Regression vs Ridge Regression") plt.xlabel("Model") plt.ylabel("Score") plt.xticks(rotation=0) plt.tight_layout() plt.show()
